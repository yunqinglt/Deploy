cmake_minimum_required(VERSION 3.22)
project(deploy_sleigh C CXX)
get_filename_component(DEPLOY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(GHIDRA_CPP "${DEPLOY_ROOT}/third_party/ghidra/Ghidra/Features/Decompiler/src/decompile/cpp")
include("${DEPLOY_ROOT}/cmake/GhidraSources.cmake")
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPLOY_ROOT}/third_party/zlib" zlib EXCLUDE_FROM_ALL)
ghidra_sources(SLEIGH_SOURCES ${GHIDRA_CORE} ${GHIDRA_SLEIGH} slgh_compile slghparse slghscan)
add_executable(deploy-sleigh ${SLEIGH_SOURCES})
target_compile_features(deploy-sleigh PRIVATE cxx_std_17)
target_include_directories(deploy-sleigh PRIVATE "${DEPLOY_ROOT}/third_party/zlib" "${CMAKE_CURRENT_BINARY_DIR}/zlib")
target_link_libraries(deploy-sleigh PRIVATE zlibstatic)
if(MSVC)
    target_compile_definitions(deploy-sleigh PRIVATE _CRT_SECURE_NO_WARNINGS _WINDOWS)
endif()
